#include<stdio.h>
void main()
{
	char str[100];
	int i,count=0;
	printf("Enter a sentence\n");
	gets(str);
	
	for(int i=0;str[i]!='\0';i++)
	{
		if(str[i]==' ')
		{
			count++;
		}
	}
	count=count+1;
	
	printf("Number of words=%d",count);
}
